cmake_minimum_required(VERSION 3.24)
project(gnni_q1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GNNI_ENABLE_CUDA "Build CUDA backends (requires a CUDA toolkit)" OFF)
include(CTest)
if(MSVC)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
endif()
set(GNNI_BUILD_TESTS ${BUILD_TESTING} CACHE BOOL "Build CPU tests" FORCE)
add_subdirectory(CPU)
add_subdirectory(common)
if(GNNI_ENABLE_CUDA)
  if(UNIX)
    string(APPEND CMAKE_CUDA_FLAGS_INIT " -U_GNU_SOURCE")
  endif()
  enable_language(CUDA)
  add_subdirectory(cuda)
endif()
add_library(q1_engine integration/engine.cpp)
target_include_directories(q1_engine PUBLIC integration)
target_link_libraries(q1_engine PUBLIC gnni_core gnni_common)
if(GNNI_ENABLE_CUDA)
  target_link_libraries(q1_engine PUBLIC gnni_cuda)
  target_compile_definitions(q1_engine PUBLIC Q1_HAS_CUDA=1)
endif()
add_executable(gnni_q1 integration/main.cpp)
target_link_libraries(gnni_q1 PRIVATE q1_engine)
if(BUILD_TESTING)
  add_executable(q1_integration_test integration/test_engine.cpp)
  target_link_libraries(q1_integration_test PRIVATE q1_engine)
  add_test(NAME q1_integration COMMAND q1_integration_test)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME q1_cli COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/integration/test_cli.py $<TARGET_FILE:gnni_q1>)
endif()
